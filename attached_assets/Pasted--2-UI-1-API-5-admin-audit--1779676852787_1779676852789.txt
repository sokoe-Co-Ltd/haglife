 POST /api/audit/cells/:cellId/skip-reason — 理由未入力のskippedセルに理由を後追い登録

各エンドポイントの仕様：
yamlUnskipRequest:
  type: object
  properties:
    restoreStatus:
      type: string
      enum: [planned, done]
      default: done
      # planned: 計画状態に戻す（未実施を取り消す）
      # done: 実施扱いに変更

SkipReasonUpdateRequest:
  type: object
  required: [skipReason]
  properties:
    skipReason:
      type: string
      minLength: 1

NoteUpdateRequest:
  type: object
  required: [note]
  properties:
    note:
      type: string
      minLength: 1
全エンドポイントで route_sheet_cell_audit_log に記録すること（action: unskip / update_skip_reason / update_note）。
lib/api-spec/openapi.yaml に追加 → codegen 実行。

tsxfunction WeekdayTabs({ weekday, date, onChange }) {

          )}

        </button>
      ))}
    </div>
  );
}

3-4. セル本体（AuditGanttCell.tsx）
時間軸に対する位置は left/width を百分率で計算：
tsxfunction AuditGanttCell({ cell }) {
  const [openModal, setOpenModal] = useState(false);
  
  // 7:00 を 0%、19:00 を 100% として時刻を百分率変換
  const left = timeToPercent(cell.actualStartTime ?? cell.plannedStartTime);
  const right = timeToPercent(cell.actualEndTime ?? cell.plannedEndTime);
  const width = right - left;
  
  // status と staffReassigned に応じた色分け
  const colorClasses = getCellColor(cell);
  
  return (
    <>
      <button
        className={cn(
          'absolute top-1.5 bottom-1.5 rounded-md p-1.5 text-xs',
          'flex flex-col justify-center overflow-hidden',
          'hover:ring-2 hover:ring-offset-1 hover:ring-foreground/20 transition',
          colorClasses
        )}
        style={{ left: `${left}%`, width: `${width}%` }}
        onClick={() => setOpenModal(true)}
      >
        <span className={cn('font-medium', cell.status === 'skipped' && 'line-through')}>
          {cell.residentName}
        </span>
        <span className="text-[10px] opacity-80">
          {cell.actualServiceLabel ?? cell.plannedServiceLabel}
          {cell.status === 'skipped' && ` ・ ${cell.skipReason || '未'}`}
          {cell.status === 'added' && ' ・追加'}
          {cell.staffReassigned && ' ・代替'}
        </span>
        
        {/* 変更メモあり：右上に黄色ドット */}
        {cell.modifiedNote && (
          <span className="absolute top-1 right-1 size-1.5 rounded-full bg-amber-500" />
        )}
        
        {/* 追加サービス：右上にベル */}
        {cell.isAdHoc && (
          <Bell className="absolute top-1 right-1 size-2.5 text-purple-700" />
        )}
        
        {/* 理由未入力のskipped：警告アイコン */}
        {cell.status === 'skipped' && cell.skipReason === '(理由未入力)' && (
          <AlertTriangle className="absolute top-1 right-1 size-2.5 text-orange-600" />
        )}
      </button>
      
      <AuditCellModal cell={cell} open={openModal} onClose={() => setOpenModal(false)} />
    </>
  );
}

function getCellColor(cell): string {
  if (cell.status === 'skipped') {
    return 'bg-red-50 border border-red-500/60 text-red-900';
  }
  if (cell.isAdHoc || cell.status === 'added') {
    return 'bg-purple-50 border border-purple-500/60 text-purple-900';
  }
  if (cell.staffReassigned) {
    return 'bg-blue-50 border border-blue-500/60 text-blue-900';
  }
  return 'bg-background border border-border/60';
}

tsxfunction AuditCellModal({ cell, open, onClose }) {
  const { data: history } = useGetAuditCellHistory(cell.id, { enabled: open });
  const deleteNote = useDeleteAuditCellNote();
  const updateNote = usePostAuditCellNote();
  const unskip = usePostAuditCellUnskip();
  const updateSkipReason = usePostAuditCellSkipReason();
  
  return (
    <Dialog open={open} onOpenChange={onClose}>
      <DialogContent className="max-w-2xl">
        <DialogHeader>
          <DialogTitle>
            {cell.residentName} ・ {cell.plannedServiceLabel ?? '追加サービス'}
          </DialogTitle>
        </DialogHeader>
        
        <div className="space-y-4">
          {/* 計画 vs 実態 */}
          <ComparisonTable cell={cell} />
          
          {/* ステータスごとのアクション */}
          {cell.status === 'skipped' && (
            <SkippedSection
              cell={cell}
              onUnskip={() => unskip.mutate({ cellId: cell.id, restoreStatus: 'done' })}
              onUpdateReason={(reason) => updateSkipReason.mutate({ cellId: cell.id, skipReason: reason })}
            />
          )}
          
          {/* 変更メモ */}
          <NoteSection
            note={cell.modifiedNote}
            onUpdate={(note) => updateNote.mutate({ cellId: cell.id, note })}
            onDelete={() => deleteNote.mutate({ cellId: cell.id })}
          />
          
          {/* 変更履歴 */}
          <HistorySection history={history} />
        </div>
      </DialogContent>
    </Dialog>
  );
}

tsxfunction NoteSection({ note, onUpdate, onDelete }) {
  const [editing, setEditing] = useState(false);
  const [text, setText] = useState(note ?? '');
  
  if (!note && !editing) {
    return (
      <button onClick={() => setEditing(true)} className="text-xs text-muted-foreground hover:underline">
        + 変更メモを追加
      </button>
    );
  }
  
  return (
    <div className="p-3 rounded-md border bg-amber-50/40 border-amber-300">
      <div className="flex items-center justify-between mb-2">
        <span className="text-sm font-medium">変更メモ</span>
        {!editing && (
          <div className="flex gap-2">
            <button onClick={() => setEditing(true)} className="text-xs">編集</button>
            <button onClick={onDelete} className="text-xs text-red-700">消去</button>
          </div>
        )}
      </div>
      {editing ? (
        <div className="space-y-2">
          <textarea
            value={text}
            onChange={(e) => setText(e.target.value)}
            className="w-full text-sm border rounded p-2"
            rows={3}
          />
          <div className="flex gap-2 justify-end">
            <button onClick={() => { setText(note ?? ''); setEditing(false); }} className="text-xs">キャンセル</button>
            <button
              onClick={() => { onUpdate(text); setEditing(false); }}
              className="text-xs bg-foreground text-background px-2 py-1 rounded"
            >
              保存
            </button>
          </div>
        </div>
      ) : (
        <p className="text-sm whitespace-pre-wrap">{note}</p>
      )}
    </div>
  );
}

function actionLabel(action: string): string {
  return {
    create: '作成',
    update: '更新',
    delete: '削除',

5-2. 採用ダイアログ（AcceptDialog.tsx）
tsxfunction AcceptDialog({ notification, onClose, onSuccess }) {
  const [scope, setScope] = useState<'day_only' | 'add_to_template'>('day_only');
  const accept = usePostAuditNotificationAccept();
  const date = useDateContext(); // 親から渡す or useParams
  const weekday = new Date(date).getDay();
  
  if (!notification) return null;
  
  const handleSubmit = () => {
    accept.mutate(
      {
        cellId: notification.cellId,
        scope,
        weekday: scope === 'add_to_template' ? weekday : undefined,
      },
      { onSuccess },
    );
  };
  
  return (
    <Dialog open={!!notification} onOpenChange={onClose}>
      <DialogContent>
        <DialogHeader>
          <DialogTitle>追加サービスの採用</DialogTitle>
          <DialogDescription>
            {notification.residentName} ・ {notification.actualServiceLabel} を採用します。
          </DialogDescription>
        </DialogHeader>
        
        <RadioGroup value={scope} onValueChange={setScope}>
          <div className="flex items-start gap-2">
            <RadioGroupItem value="day_only" id="day_only" />
            <label htmlFor="day_only" className="text-sm">
              <strong>その日限り</strong>の実績として確定
              <p className="text-xs text-muted-foreground mt-0.5">
                通知から消えます。テンプレートには影響しません。
              </p>
            </label>
          </div>
          <div className="flex items-start gap-2">
            <RadioGroupItem value="add_to_template" id="template" />
            <label htmlFor="template" className="text-sm">
              <strong>曜日テンプレートに追加</strong>
              <p className="text-xs text-muted-foreground mt-0.5">
                次回以降の{['日','月','火','水','木','金','土'][weekday]}曜日にも反映されます。
              </p>
            </label>
          </div>
        </RadioGroup>
        
        <DialogFooter>
          <button onClick={onClose} className="text-sm px-3 py-1.5 border rounded">キャンセル</button>
          <button onClick={handleSubmit} className="text-sm px-3 py-1.5 bg-foreground text-background rounded">
            採用する
          </button>
        </DialogFooter>
      </DialogContent>
    </Dialog>
  );
}
